// Runs one motor. speed: -255 (full reverse) to +255 (full forward). Given.
void motor(int in1, int in2, int pwm, int speed) {
  digitalWrite(in1, speed > 0 ? HIGH : LOW);
  digitalWrite(in2, speed < 0 ? HIGH : LOW);
  analogWrite(pwm, abs(speed));
}
