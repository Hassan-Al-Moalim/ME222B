// ===================== 6. MOTORS: TWO H-BRIDGES (Lab 1 wiring) =====================
// One dual H-bridge per side, as in Lab 1.
// FWD = that side's front motor, BWD = its rear motor.
// Copy YOUR pins from your Lab 1 table. Do not use GPIO 0, 4, 16, 17, 21, 22
// (BOOT button, lidar, IMU).

// LEFT driver
#define L_FWD_IN1   26
#define L_FWD_IN2   27
#define L_FWD_PWM   25
#define L_BWD_IN1   13
#define L_BWD_IN2   18
#define L_BWD_PWM   19

// RIGHT driver
#define R_FWD_IN1   14
#define R_FWD_IN2   32
#define R_FWD_PWM   33
#define R_BWD_IN1   5
#define R_BWD_IN2   15
#define R_BWD_PWM   23

// In Lab 1 you may have swapped HIGH/LOW for a motor to make it turn
// forward. Here, set that motor's polarity to -1 instead.
const int L_FWD_POL = 1, L_BWD_POL = 1;
const int R_FWD_POL = 1, R_BWD_POL = 1;

const int N_CH = 4;
const int CH_EN[N_CH]   = {L_FWD_PWM, L_BWD_PWM, R_FWD_PWM, R_BWD_PWM};
const int CH_A[N_CH]    = {L_FWD_IN1, L_BWD_IN1, R_FWD_IN1, R_BWD_IN1};
const int CH_B[N_CH]    = {L_FWD_IN2, L_BWD_IN2, R_FWD_IN2, R_BWD_IN2};
const int CH_POL[N_CH]  = {L_FWD_POL, L_BWD_POL, R_FWD_POL, R_BWD_POL};
const int CH_SIDE[N_CH] = {0, 0, 1, 1};     // 0 = left, 1 = right

int chDir[N_CH] = {0, 0, 0, 0};
uint32_t chChangedMs[N_CH] = {0, 0, 0, 0};
int leftOut = 0, rightOut = 0;              // last command, -255..255

// one motor, cmd -255..255
void runChannel(int ch, int cmd) {
  int d = cmd > 0 ? 1 : (cmd < 0 ? -1 : chDir[ch]);
  if (d != chDir[ch]) {                     // short pause before reversing
    analogWrite(CH_EN[ch], 0);
    bool fwd = d * CH_POL[ch] > 0;
    digitalWrite(CH_A[ch], fwd ? HIGH : LOW);
    digitalWrite(CH_B[ch], fwd ? LOW : HIGH);
    chDir[ch] = d;
    chChangedMs[ch] = millis();
  }
  if (millis() - chChangedMs[ch] < 20) cmd = 0;
  analogWrite(CH_EN[ch], abs(cmd));
}

// left, right: -255 (full reverse) to +255 (full forward)
void setMotors(int left, int right) {
  left = constrain(left, -255, 255);
  right = constrain(right, -255, 255);
  for (int ch = 0; ch < N_CH; ++ch) runChannel(ch, CH_SIDE[ch] == 0 ? left : right);
  leftOut = left; rightOut = right;
}

void stopMotors() { setMotors(0, 0); }

void motorsBegin() {
  for (int ch = 0; ch < N_CH; ++ch) {
    pinMode(CH_EN[ch], OUTPUT); pinMode(CH_A[ch], OUTPUT); pinMode(CH_B[ch], OUTPUT);
    digitalWrite(CH_A[ch], LOW); digitalWrite(CH_B[ch], LOW);
    analogWrite(CH_EN[ch], 0);
  }
}
